import Obsmetrologia.Language.Positing.Four
import Obsmetrologia.Language.Positing.Value.PreCategory

/-!
# Record — the counting value carrier under the charge

The record sector: a value is a pair of tallies — how many times a cell has
been told true, how many times told false — rather than a pair of flags
(corpus: `content/.../sheaf`, `content/.../inconsistency-sheaf`; the charge is
the record's shadow under "is the tally positive?"). The carrier is the twist
product of the unbounded lattice `(ℕ, min, max)` with itself, carrying the
same two orders as `Charge`: knowledge `≤ₖ` (both tallies increasing) and
truth `≤ₜ` (positive up, negative down).

## Main definitions

* `Record` — the pair of tallies; `kBot` the empty record.
* `joinK`, `meetK`, `joinT`, `meetT` — the componentwise `max`/`min` lattice
  operations of the two orders.
* `add` — the record merge: pointwise addition, carrying multiplicity.
* `charge` — the readout `Record → Charge` by tally positivity.
* `topMass` — the contradiction mass `min pos neg`.
* `succPos`, `iter` — the tally-increment dynamic and finite iteration.

## Main results

* `record_interlaced` — both orders are lattices and each operation is
  monotone in the other order: the interlaced laws, in the algebraic form of
  `InterlacedBilattice`'s fields.
* `no_kTop`, `no_tTop`, `no_tBot` — three of the four bounds do not exist:
  the carrier fails `InterlacedBilattice`'s boundedness, so `four_initial`
  does not apply to it. Together with `InterlacedBilattice.joinK_kTop` this
  places `⊤` as an artifact of demanding a bound.
* `record_escapes_strict_finite` — `succPos` is `≤ₖ`-inflationary and
  monotone yet has no finite settling stage: `iter succPos (k+1) r ≠
  iter succPos k r` for every `k`. `Charge.settles_by_three` fails on this
  carrier; the strict-finite regime is a boundedness artifact, and the
  `κ = ω` regime is inhabited. `joinClosure_one_step` is the counterpoint:
  every join-bounded closure still settles in one step.
* `charge_joinK`/`meetK`/`joinT`/`meetT` — the readout is a homomorphism for
  all four operations; `charge_surjective`, `charge_not_injective` — it is a
  proper collapse.
* `charge_add_eq_charge_joinK` — the readout cannot distinguish the counting
  merge from the idempotent join.
* `topMass_add` — the counting merge is superadditive in contradiction mass:
  merging never destroys held contradiction. (The idempotent `Charge` join
  collapses it: `joinK ⊤ ⊤ = ⊤`.)
* `bottom_reflecting_hom_eq_charge` — the collapse is forced: any map
  `Record → Charge` preserving the four operations, sending the empty record
  to `⊥ₖ`, and reflecting `⊥ₖ` is the charge readout, pointwise.

## References

Fitting's twist product `L ⊙ L`, here over the unbounded lattice
`(ℕ, min, max)`; sibling `Calculus.Logic.TwistProductResiduation` builds the
bounded case over a residuated lattice.

## Tags

record, bilattice, twist product, unbounded, charge, multiplicity
-/

namespace Obsmetrologia

/-- A record: the pair of evidence tallies for one cell — told-true count and
told-false count. The charge is this pair's positivity shadow. -/
structure Record where
  /-- The told-true tally. -/
  pos : Nat
  /-- The told-false tally. -/
  neg : Nat
  deriving DecidableEq, Repr, Inhabited

namespace Record

/-- Records are equal when their two tallies are. -/
theorem ext {a b : Record} (hp : a.pos = b.pos) (hn : a.neg = b.neg) : a = b := by
  cases a; cases b; cases hp; cases hn; rfl

/-- The empty record — no evidence either way; the knowledge bottom. -/
def kBot : Record := ⟨0, 0⟩

/-! ## The six operations -/

/-- Joining-knowledge `∨ₖ` — componentwise `max`. -/
def joinK (a b : Record) : Record := ⟨max a.pos b.pos, max a.neg b.neg⟩
/-- Meeting-knowledge `∧ₖ` — componentwise `min`. -/
def meetK (a b : Record) : Record := ⟨min a.pos b.pos, min a.neg b.neg⟩
/-- Joining-truth `∨ₜ` — positive `max`, negative `min`. -/
def joinT (a b : Record) : Record := ⟨max a.pos b.pos, min a.neg b.neg⟩
/-- Meeting-truth `∧ₜ` — positive `min`, negative `max`. -/
def meetT (a b : Record) : Record := ⟨min a.pos b.pos, max a.neg b.neg⟩
/-- The record merge: pointwise addition. Unlike the joins it is not
idempotent — it carries multiplicity. -/
def add (a b : Record) : Record := ⟨a.pos + b.pos, a.neg + b.neg⟩

/-! ## The two orders -/

/-- The knowledge order: both tallies increase. -/
def KLE (a b : Record) : Prop := a.pos ≤ b.pos ∧ a.neg ≤ b.neg
/-- The truth order: positive tally up, negative tally down. -/
def TLE (a b : Record) : Prop := a.pos ≤ b.pos ∧ b.neg ≤ a.neg

theorem kle_refl (a : Record) : KLE a a := ⟨Nat.le_refl _, Nat.le_refl _⟩
theorem kle_trans (a b c : Record) : KLE a b → KLE b c → KLE a c := by
  intro h1 h2; exact ⟨Nat.le_trans h1.1 h2.1, Nat.le_trans h1.2 h2.2⟩
theorem kle_antisymm (a b : Record) : KLE a b → KLE b a → a = b := by
  intro h1 h2; exact ext (Nat.le_antisymm h1.1 h2.1) (Nat.le_antisymm h1.2 h2.2)

theorem tle_refl (a : Record) : TLE a a := ⟨Nat.le_refl _, Nat.le_refl _⟩
theorem tle_trans (a b c : Record) : TLE a b → TLE b c → TLE a c := by
  intro h1 h2; exact ⟨Nat.le_trans h1.1 h2.1, Nat.le_trans h2.2 h1.2⟩
theorem tle_antisymm (a b : Record) : TLE a b → TLE b a → a = b := by
  intro h1 h2; exact ext (Nat.le_antisymm h1.1 h2.1) (Nat.le_antisymm h2.2 h1.2)

theorem le_joinK_left (a b : Record) : KLE a (joinK a b) :=
  ⟨Nat.le_max_left _ _, Nat.le_max_left _ _⟩
theorem le_joinK_right (a b : Record) : KLE b (joinK a b) :=
  ⟨Nat.le_max_right _ _, Nat.le_max_right _ _⟩
theorem joinK_le (a b c : Record) : KLE a c → KLE b c → KLE (joinK a b) c := by
  intro h1 h2
  exact ⟨Nat.max_le.mpr ⟨h1.1, h2.1⟩, Nat.max_le.mpr ⟨h1.2, h2.2⟩⟩
theorem meetK_le_left (a b : Record) : KLE (meetK a b) a :=
  ⟨Nat.min_le_left _ _, Nat.min_le_left _ _⟩
theorem meetK_le_right (a b : Record) : KLE (meetK a b) b :=
  ⟨Nat.min_le_right _ _, Nat.min_le_right _ _⟩
theorem le_meetK (a b c : Record) : KLE c a → KLE c b → KLE c (meetK a b) := by
  intro h1 h2
  exact ⟨Nat.le_min.mpr ⟨h1.1, h2.1⟩, Nat.le_min.mpr ⟨h1.2, h2.2⟩⟩

/-- The empty record is the knowledge-least record. -/
theorem kBot_le (a : Record) : KLE kBot a := ⟨Nat.zero_le _, Nat.zero_le _⟩

/-! ## The interlaced laws (the algebraic field shapes of
`InterlacedBilattice`, minus the missing bounds) -/

theorem joinK_comm (a b : Record) : joinK a b = joinK b a :=
  ext (Nat.max_comm _ _) (Nat.max_comm _ _)
theorem joinK_assoc (a b c : Record) : joinK (joinK a b) c = joinK a (joinK b c) :=
  ext (Nat.max_assoc _ _ _) (Nat.max_assoc _ _ _)
theorem meetK_comm (a b : Record) : meetK a b = meetK b a :=
  ext (Nat.min_comm _ _) (Nat.min_comm _ _)
theorem absorbK_join (a b : Record) : joinK a (meetK a b) = a :=
  ext (Nat.max_eq_left (Nat.min_le_left _ _)) (Nat.max_eq_left (Nat.min_le_left _ _))
theorem absorbK_meet (a b : Record) : meetK a (joinK a b) = a :=
  ext (Nat.min_eq_left (Nat.le_max_left _ _)) (Nat.min_eq_left (Nat.le_max_left _ _))
theorem joinK_kBot (a : Record) : joinK kBot a = a :=
  ext (Nat.zero_max _) (Nat.zero_max _)
theorem joinT_comm (a b : Record) : joinT a b = joinT b a :=
  ext (Nat.max_comm _ _) (Nat.min_comm _ _)
theorem joinT_assoc (a b c : Record) : joinT (joinT a b) c = joinT a (joinT b c) :=
  ext (Nat.max_assoc _ _ _) (Nat.min_assoc _ _ _)
theorem meetT_comm (a b : Record) : meetT a b = meetT b a :=
  ext (Nat.min_comm _ _) (Nat.max_comm _ _)
theorem absorbT_join (a b : Record) : joinT a (meetT a b) = a :=
  ext (Nat.max_eq_left (Nat.min_le_left _ _)) (Nat.min_eq_left (Nat.le_max_left _ _))
theorem absorbT_meet (a b : Record) : meetT a (joinT a b) = a :=
  ext (Nat.min_eq_left (Nat.le_max_left _ _)) (Nat.max_eq_left (Nat.min_le_left _ _))

/-- Truth meet is knowledge-monotone (interlacing, in algebraic form). -/
theorem meetT_kmono (a b c : Record) (h : joinK a b = b) :
    joinK (meetT a c) (meetT b c) = meetT b c := by
  have hp := congrArg Record.pos h
  have hn := congrArg Record.neg h
  simp only [joinK] at hp hn
  apply ext <;> simp only [joinK, meetT] <;> omega

/-- Truth join is knowledge-monotone. -/
theorem joinT_kmono (a b c : Record) (h : joinK a b = b) :
    joinK (joinT a c) (joinT b c) = joinT b c := by
  have hp := congrArg Record.pos h
  have hn := congrArg Record.neg h
  simp only [joinK] at hp hn
  apply ext <;> simp only [joinK, joinT] <;> omega

/-- Knowledge meet is truth-monotone. -/
theorem meetK_tmono (a b c : Record) (h : joinT a b = b) :
    joinT (meetK a c) (meetK b c) = meetK b c := by
  have hp := congrArg Record.pos h
  have hn := congrArg Record.neg h
  simp only [joinT] at hp hn
  apply ext <;> simp only [joinT, meetK] <;> omega

/-- Knowledge join is truth-monotone. -/
theorem joinK_tmono (a b c : Record) (h : joinT a b = b) :
    joinT (joinK a c) (joinK b c) = joinK b c := by
  have hp := congrArg Record.pos h
  have hn := congrArg Record.neg h
  simp only [joinT] at hp hn
  apply ext <;> simp only [joinT, joinK] <;> omega

/-- The interlaced-law bundle: both orders are lattices with the knowledge
bottom, and each operation is monotone in the other order — every
`InterlacedBilattice` field shape that does not mention `kTop`, `tBot`, or
`tTop` holds on records. -/
theorem record_interlaced :
    (∀ a b, joinK a b = joinK b a) ∧ (∀ a b, meetT a b = meetT b a)
    ∧ (∀ a b, joinK a (meetK a b) = a) ∧ (∀ a b, joinT a (meetT a b) = a)
    ∧ (∀ a, joinK kBot a = a)
    ∧ (∀ a b c, joinK a b = b → joinK (meetT a c) (meetT b c) = meetT b c)
    ∧ (∀ a b c, joinT a b = b → joinT (meetK a c) (meetK b c) = meetK b c) :=
  ⟨joinK_comm, meetT_comm, absorbK_join, absorbT_join, joinK_kBot,
    meetT_kmono, meetK_tmono⟩

/-! ## The missing bounds -/

/-- **There is no knowledge top.** No record absorbs the knowledge join: the
told-true tally can always be exceeded. `Record` therefore fails the
boundedness hypothesis of `InterlacedBilattice`, and `four_initial` does not
apply to it: held `⊤` is what a carrier buys by demanding a bound. -/
theorem no_kTop : ¬ ∃ t : Record, ∀ r, KLE r t := by
  intro h
  cases h with
  | intro t ht =>
    exact Nat.not_succ_le_self t.pos ((ht ⟨t.pos + 1, 0⟩).1)

/-- There is no truth top: the told-true tally is unbounded above in `≤ₜ`. -/
theorem no_tTop : ¬ ∃ t : Record, ∀ r, TLE r t := by
  intro h
  cases h with
  | intro t ht =>
    exact Nat.not_succ_le_self t.pos ((ht ⟨t.pos + 1, 0⟩).1)

/-- There is no truth bottom: the told-false tally is unbounded above, and
`≤ₜ` reads it downward. -/
theorem no_tBot : ¬ ∃ f : Record, ∀ r, TLE f r := by
  intro h
  cases h with
  | intro f hf =>
    exact Nat.not_succ_le_self f.neg ((hf ⟨0, f.neg + 1⟩).2)

/-! ## The escape from the strict-finite regime -/

/-- Finite iteration of an endomap on records. -/
def iter (f : Record → Record) : Nat → Record → Record
  | 0, r => r
  | k + 1, r => f (iter f k r)

/-- The tally-increment dynamic: one more told-true. -/
def succPos (r : Record) : Record := ⟨r.pos + 1, r.neg⟩

theorem succPos_infl (r : Record) : KLE r (succPos r) :=
  ⟨Nat.le_succ _, Nat.le_refl _⟩

theorem succPos_mono (a b : Record) (h : KLE a b) : KLE (succPos a) (succPos b) :=
  ⟨Nat.succ_le_succ h.1, h.2⟩

/-- The told-true tally after `k` steps. -/
theorem iter_succPos_pos (k : Nat) (r : Record) :
    (iter succPos k r).pos = r.pos + k := by
  induction k with
  | zero => rfl
  | succ k ih =>
    show (iter succPos k r).pos + 1 = r.pos + (k + 1)
    rw [ih]
    rfl

/-- **No finite settling stage.** The increment dynamic never revisits: every
stage differs from the next, at every seed. -/
theorem iter_succPos_ne (k : Nat) (r : Record) :
    iter succPos (k + 1) r ≠ iter succPos k r := by
  intro h
  have h2 := congrArg Record.pos h
  rw [iter_succPos_pos, iter_succPos_pos] at h2
  exact Nat.succ_ne_self (r.pos + k) h2

/-- **The record carrier escapes the strict-finite regime.** `succPos` is
`≤ₖ`-inflationary and monotone, yet no iterate is a settling stage —
`Charge.settles_by_three` (every inflationary endomap on the four-element
carrier settles by stage 3) fails on records. The `κ = ω` regime is
inhabited by a derived carrier, not by asserted regime data. -/
theorem record_escapes_strict_finite :
    (∀ r, KLE r (succPos r))
    ∧ (∀ a b, KLE a b → KLE (succPos a) (succPos b))
    ∧ (∀ k r, iter succPos (k + 1) r ≠ iter succPos k r) :=
  ⟨succPos_infl, succPos_mono, iter_succPos_ne⟩

/-- The counterpoint: a join-bounded closure on records still settles in one
step — `joinK k` is idempotent. Unboundedness of the carrier, not failure of
the closure laws, is what admits the escaping dynamic. -/
theorem joinClosure_one_step (k r : Record) :
    joinK k (joinK k r) = joinK k r := by
  apply ext
  · show max k.pos (max k.pos r.pos) = max k.pos r.pos
    rw [← Nat.max_assoc, Nat.max_self]
  · show max k.neg (max k.neg r.neg) = max k.neg r.neg
    rw [← Nat.max_assoc, Nat.max_self]

/-! ## The charge readout -/

/-- The charge of a record: the positivity of each tally. This is the map the
corpus calls the shadow of the record sector — `Record`'s two counters
flattened to `Charge`'s two flags. -/
def charge (r : Record) : Charge := Charge.ofSlots (r.pos != 0) (r.neg != 0)

theorem bne_zero_max (p q : Nat) : ((max p q) != 0) = ((p != 0) || (q != 0)) := by
  cases p <;> cases q <;> simp [Nat.succ_max_succ]

theorem bne_zero_min (p q : Nat) : ((min p q) != 0) = ((p != 0) && (q != 0)) := by
  cases p <;> cases q <;> simp [Nat.succ_min_succ]

theorem bne_zero_add (p q : Nat) : ((p + q) != 0) = ((p != 0) || (q != 0)) := by
  cases p <;> cases q <;> simp

@[simp] theorem charge_kBot : charge kBot = Charge.kBot := rfl

/-- The readout preserves the knowledge join. -/
theorem charge_joinK (a b : Record) :
    charge (joinK a b) = Charge.joinK (charge a) (charge b) := by
  simp [charge, joinK, Charge.joinK, bne_zero_max]

/-- The readout preserves the knowledge meet. -/
theorem charge_meetK (a b : Record) :
    charge (meetK a b) = Charge.meetK (charge a) (charge b) := by
  simp [charge, meetK, Charge.meetK, bne_zero_min]

/-- The readout preserves the truth join. -/
theorem charge_joinT (a b : Record) :
    charge (joinT a b) = Charge.joinT (charge a) (charge b) := by
  simp [charge, joinT, Charge.joinT, bne_zero_max, bne_zero_min]

/-- The readout preserves the truth meet. -/
theorem charge_meetT (a b : Record) :
    charge (meetT a b) = Charge.meetT (charge a) (charge b) := by
  simp [charge, meetT, Charge.meetT, bne_zero_max, bne_zero_min]

/-- Every charge is some record's readout. -/
theorem charge_surjective (c : Charge) : ∃ r : Record, charge r = c := by
  cases c with
  | potential => exact ⟨⟨0, 0⟩, rfl⟩
  | active => exact ⟨⟨1, 0⟩, rfl⟩
  | absurd => exact ⟨⟨1, 1⟩, rfl⟩
  | stable => exact ⟨⟨0, 1⟩, rfl⟩

/-- The readout is a proper collapse: distinct records share a charge. -/
theorem charge_not_injective :
    ∃ a b : Record, charge a = charge b ∧ a ≠ b :=
  ⟨⟨1, 0⟩, ⟨2, 0⟩, rfl, by decide⟩

/-- **The readout cannot tell the counting merge from the idempotent join.**
Under `charge`, `add` and `joinK` are the same operation — multiplicity is
exactly what the shadow forgets. -/
theorem charge_add_eq_charge_joinK (a b : Record) :
    charge (add a b) = charge (joinK a b) := by
  simp [charge, add, joinK, bne_zero_add, bne_zero_max]

/-! ## Contradiction mass -/

/-- The contradiction mass of a record: the number of matched
told-true/told-false pairs it holds. -/
def topMass (r : Record) : Nat := min r.pos r.neg

/-- A record reads out as held `⊤` exactly when it holds positive
contradiction mass. -/
theorem charge_eq_kTop_iff (r : Record) :
    charge r = Charge.kTop ↔ 0 < topMass r := by
  cases r with
  | mk p n =>
    cases p <;> cases n <;> simp [charge, topMass] <;> decide

theorem topMass_le_pos (r : Record) : topMass r ≤ r.pos := by
  show min r.pos r.neg ≤ r.pos
  rw [Nat.min_def]
  split
  · exact Nat.le_refl _
  · next h => exact Nat.le_of_succ_le (Nat.gt_of_not_le h)

theorem topMass_le_neg (r : Record) : topMass r ≤ r.neg := by
  show min r.pos r.neg ≤ r.neg
  rw [Nat.min_def]
  split
  · next h => exact h
  · exact Nat.le_refl _

theorem neg_le_topMass (r : Record) (h : r.neg ≤ r.pos) : r.neg ≤ topMass r := by
  show r.neg ≤ min r.pos r.neg
  by_cases hc : r.pos ≤ r.neg
  · rw [Nat.min_def, if_pos hc]
    exact h
  · rw [Nat.min_def, if_neg hc]
    exact Nat.le_refl _

theorem pos_le_topMass (r : Record) (h : r.pos ≤ r.neg) : r.pos ≤ topMass r := by
  show r.pos ≤ min r.pos r.neg
  rw [Nat.min_def, if_pos h]
  exact Nat.le_refl _

/-- **The counting merge never destroys contradiction mass** — it is
superadditive. The idempotent `Charge` join collapses mass
(`joinK ⊤ ⊤ = ⊤`); the record merge carries it. -/
theorem topMass_add (a b : Record) :
    topMass a + topMass b ≤ topMass (add a b) :=
  Nat.le_min.mpr
    ⟨Nat.add_le_add (Nat.min_le_left _ _) (Nat.min_le_left _ _),
      Nat.add_le_add (Nat.min_le_right _ _) (Nat.min_le_right _ _)⟩

/-! ## The collapse is forced -/

/-- **Any bottom-reflecting homomorphic readout is the charge.** A map
`Record → Charge` that preserves the four lattice operations, sends the empty
record to `⊥ₖ`, and sends only the empty record to `⊥ₖ` agrees with `charge`
pointwise. The Belnap flags are the only bottom-faithful four-valued shadow
of the tallies; the lossiness measured by `charge_not_injective` and
`charge_add_eq_charge_joinK` is not a design choice in the readout but a
property of reading records four-valuedly at all. -/
theorem bottom_reflecting_hom_eq_charge (h : Record → Charge)
    (hJK : ∀ a b, h (joinK a b) = Charge.joinK (h a) (h b))
    (hMK : ∀ a b, h (meetK a b) = Charge.meetK (h a) (h b))
    (hJT : ∀ a b, h (joinT a b) = Charge.joinT (h a) (h b))
    (hMT : ∀ a b, h (meetT a b) = Charge.meetT (h a) (h b))
    (hbot : h kBot = Charge.kBot)
    (hrefl : ∀ r, h r = Charge.kBot → r = kBot) :
    ∀ r, h r = charge r := by
  -- the one-tick records are forced
  have hone : h ⟨1, 0⟩ = Charge.tTop := by
    have hkey := hMT ⟨1, 0⟩ kBot
    rw [show meetT ⟨1, 0⟩ kBot = kBot from rfl, hbot] at hkey
    cases hx : h ⟨1, 0⟩ <;> rw [hx] at hkey
    · exact absurd (hrefl _ hx) (by decide)
    · rfl
    · exact absurd hkey (by decide)
    · exact absurd hkey (by decide)
  have honeneg : h ⟨0, 1⟩ = Charge.tBot := by
    have hkey := hJT ⟨0, 1⟩ kBot
    rw [show joinT ⟨0, 1⟩ kBot = kBot from rfl, hbot] at hkey
    cases hx : h ⟨0, 1⟩ <;> rw [hx] at hkey
    · exact absurd (hrefl _ hx) (by decide)
    · exact absurd hkey (by decide)
    · exact absurd hkey (by decide)
    · rfl
  -- pure-positive and pure-negative records are forced
  have hpos : ∀ p, h ⟨p + 1, 0⟩ = Charge.tTop := by
    intro p
    have k1 := hMK ⟨p + 1, 0⟩ ⟨1, 0⟩
    have e1 : meetK ⟨p + 1, 0⟩ ⟨1, 0⟩ = (⟨1, 0⟩ : Record) := by
      apply ext
      · exact Nat.min_eq_right (Nat.succ_le_succ (Nat.zero_le p))
      · rfl
    rw [e1, hone] at k1
    have k2 := hMK ⟨p + 1, 0⟩ ⟨0, 1⟩
    rw [show meetK ⟨p + 1, 0⟩ ⟨0, 1⟩ = kBot from rfl, hbot, honeneg] at k2
    cases hx : h ⟨p + 1, 0⟩ <;> rw [hx] at k1 k2
    · exact absurd k1 (by decide)
    · rfl
    · exact absurd k2 (by decide)
    · exact absurd k1 (by decide)
  have hneg : ∀ n, h ⟨0, n + 1⟩ = Charge.tBot := by
    intro n
    have k1 := hMK ⟨0, n + 1⟩ ⟨0, 1⟩
    have e1 : meetK ⟨0, n + 1⟩ ⟨0, 1⟩ = (⟨0, 1⟩ : Record) := by
      apply ext
      · rfl
      · exact Nat.min_eq_right (Nat.succ_le_succ (Nat.zero_le n))
    rw [e1, honeneg] at k1
    have k2 := hMK ⟨0, n + 1⟩ ⟨1, 0⟩
    rw [show meetK ⟨0, n + 1⟩ ⟨1, 0⟩ = kBot from rfl, hbot, hone] at k2
    cases hx : h ⟨0, n + 1⟩ <;> rw [hx] at k1 k2
    · exact absurd k1 (by decide)
    · exact absurd k2 (by decide)
    · exact absurd k2 (by decide)
    · rfl
  -- mixed records are the join of the forced pieces
  intro r
  cases r with
  | mk p n =>
    cases p with
    | zero =>
      cases n with
      | zero => exact hbot
      | succ n => exact hneg n
    | succ p =>
      cases n with
      | zero => exact hpos p
      | succ n =>
        have e : joinK ⟨p + 1, 0⟩ ⟨0, n + 1⟩ = (⟨p + 1, n + 1⟩ : Record) := rfl
        have k := hJK ⟨p + 1, 0⟩ ⟨0, n + 1⟩
        rw [e, hpos p, hneg n] at k
        exact k.trans rfl

/-! ## Why an additive coboundary cannot close over Record coefficients -/

/-- The additive degree-0-to-degree-1 coboundary: a 0-cochain's pairwise sum, `add` standing in
for the classical subtraction (`add` is all `Record` offers — no inverse exists,
`no_kTop`/`no_tTop`/`no_tBot`). -/
def deltaAdd0 {M : Type} (f : M → Record) : M → M → Record := fun i j => add (f i) (f j)

/-- The additive degree-1-to-degree-2 coboundary: a 1-cochain's three-edge sum around a triangle,
again by `add`. -/
def deltaAdd1 {M : Type} (g : M → M → Record) : M → M → M → Record :=
  fun i j k => add (add (g i j) (g j k)) (g i k)

theorem deltaAdd_pos {M : Type} (f : M → Record) (i j k : M) :
    (deltaAdd1 (deltaAdd0 f) i j k).pos
      = (f i).pos + (f i).pos + ((f j).pos + (f j).pos) + ((f k).pos + (f k).pos) := by
  show ((f i).pos + (f j).pos + ((f j).pos + (f k).pos) + ((f i).pos + (f k).pos)) = _
  omega

theorem deltaAdd_neg {M : Type} (f : M → Record) (i j k : M) :
    (deltaAdd1 (deltaAdd0 f) i j k).neg
      = (f i).neg + (f i).neg + ((f j).neg + (f j).neg) + ((f k).neg + (f k).neg) := by
  show ((f i).neg + (f j).neg + ((f j).neg + (f k).neg) + ((f i).neg + (f k).neg)) = _
  omega

theorem kBot_pos : kBot.pos = 0 := rfl
theorem kBot_neg : kBot.neg = 0 := rfl

/-- **The additive coboundary closes at a triangle exactly when every vertex is `kBot`.** Every
value the composite reads is doubled and summed (`deltaAdd_pos`/`deltaAdd_neg`), and `Nat`
addition has no zero divisors past `0`, so `δ¹δ⁰f` vanishes at `(i,j,k)` iff `f i`, `f j`, `f k`
are all `kBot`. Unlike the classical alternating-sum coboundary, no cancellation is available to
make `δ¹δ⁰f` vanish on a nontrivial input: this is not a missing proof of `δ∘δ = 0`, it is a proof
that the naive additive coboundary never satisfies it except degenerately. -/
theorem deltaAdd_eq_kBot_iff {M : Type} (f : M → Record) (i j k : M) :
    deltaAdd1 (deltaAdd0 f) i j k = kBot
    ↔ (f i = kBot ∧ f j = kBot ∧ f k = kBot) := by
  constructor
  · intro h
    have hp := deltaAdd_pos f i j k
    have hn := deltaAdd_neg f i j k
    rw [h, kBot_pos] at hp
    rw [h, kBot_neg] at hn
    refine ⟨ext ?_ ?_, ext ?_ ?_, ext ?_ ?_⟩ <;>
      first | rw [kBot_pos]; omega | rw [kBot_neg]; omega
  · rintro ⟨hi, hj, hk⟩
    have hp := deltaAdd_pos f i j k
    have hn := deltaAdd_neg f i j k
    rw [hi, hj, hk] at hp hn
    exact ext (by rw [hp]; rfl) (by rw [hn]; rfl)

/-- **The additive coboundary never closes on a nontrivial cochain.** If `f` is nonzero anywhere
among `i, j, k`, the composite `δ¹δ⁰f` is nonzero at `(i, j, k)` — the direct corollary of
`deltaAdd_eq_kBot_iff` stated as the impossibility result it is: a genuine `δ∘δ = kBot`
coboundary theory over `Record` coefficients cannot be built by naively replacing subtraction
with `add` in the classical alternating-sum construction, for any choice of which faces to sum.
The obstruction traces to the same fact as `no_kTop`/`no_tTop`/`no_tBot`: `Record` has no
subtraction to cancel repeated terms. A genuine Record-coefficient coboundary theory, if one
exists, must be order-valued (matching `Closure.Odometer.CylinderDatum.GluesTo`'s `≤ₖ`-gluing,
not `=`-coboundary triviality) rather than equational. -/
theorem deltaAdd_ne_kBot_of_ne {M : Type} (f : M → Record) (i j k : M)
    (h : f i ≠ kBot ∨ f j ≠ kBot ∨ f k ≠ kBot) :
    deltaAdd1 (deltaAdd0 f) i j k ≠ kBot := by
  intro hz
  obtain ⟨hi, hj, hk⟩ := (deltaAdd_eq_kBot_iff f i j k).mp hz
  rcases h with h | h | h
  · exact h hi
  · exact h hj
  · exact h hk

end Record

/-! ## The completed record — restoring the bounds at infinity -/

/-- A completed record: extended tallies, where `none` is the point at
infinity ω. Adjoining ω to each axis restores every bound the record lacks;
the knowledge top is the double infinity — told both, unboundedly. -/
structure ORecord where
  /-- The told-true tally, possibly infinite. -/
  pos : Option Nat
  /-- The told-false tally, possibly infinite. -/
  neg : Option Nat
  deriving DecidableEq, Repr, Inhabited

namespace ORecord

/-- Completed records are equal when their two tallies are. -/
theorem ext {a b : ORecord} (hp : a.pos = b.pos) (hn : a.neg = b.neg) : a = b := by
  cases a; cases b; cases hp; cases hn; rfl

/-- Extended-tally max: ω absorbs. -/
def omax : Option Nat → Option Nat → Option Nat
  | none, _ => none
  | _, none => none
  | some a, some b => some (max a b)

/-- Extended-tally min: ω is the identity. -/
def omin : Option Nat → Option Nat → Option Nat
  | none, b => b
  | a, none => a
  | some a, some b => some (min a b)

/-- Extended-tally order: ω is the top. -/
def ole : Option Nat → Option Nat → Prop
  | _, none => True
  | none, some _ => False
  | some x, some y => x ≤ y

/-! ### Extended-tally lattice laws -/

theorem omax_comm (a b : Option Nat) : omax a b = omax b a := by
  cases a <;> cases b <;> simp only [omax, Option.some.injEq] <;> omega

theorem omax_assoc (a b c : Option Nat) : omax (omax a b) c = omax a (omax b c) := by
  cases a <;> cases b <;> cases c <;> simp only [omax, Option.some.injEq] <;> omega

theorem omin_comm (a b : Option Nat) : omin a b = omin b a := by
  cases a <;> cases b <;> simp only [omin, Option.some.injEq] <;> omega

theorem omin_assoc (a b c : Option Nat) : omin (omin a b) c = omin a (omin b c) := by
  cases a <;> cases b <;> cases c <;> simp only [omin, Option.some.injEq] <;> omega

theorem omax_absorb (a b : Option Nat) : omax a (omin a b) = a := by
  cases a <;> cases b <;> simp only [omax, omin, Option.some.injEq] <;> omega

theorem omin_absorb (a b : Option Nat) : omin a (omax a b) = a := by
  cases a <;> cases b <;> simp only [omax, omin, Option.some.injEq] <;> omega

theorem omax_zero (a : Option Nat) : omax (some 0) a = a := by
  cases a <;> simp only [omax, Option.some.injEq] <;> omega

theorem omin_omega (a : Option Nat) : omin none a = a := rfl

/-- Cross-monotonicity of `omin` against an `omax`-order hypothesis. -/
theorem omin_cross (x y z : Option Nat) (h : omax x y = y) :
    omax (omin x z) (omin y z) = omin y z := by
  cases x <;> cases y <;> cases z <;>
    (try simp [omax, omin] at h ⊢) <;> omega

/-- Cross-monotonicity of `omax` against an `omax`-order hypothesis. -/
theorem omax_cross (x y z : Option Nat) (h : omax x y = y) :
    omax (omax x z) (omax y z) = omax y z := by
  cases x <;> cases y <;> cases z <;>
    (try simp [omax] at h ⊢) <;> omega

/-- Cross-monotonicity of `omin` against an `omin`-order hypothesis. -/
theorem omin_cross' (x y z : Option Nat) (h : omin x y = y) :
    omin (omin x z) (omin y z) = omin y z := by
  cases x <;> cases y <;> cases z <;>
    (try simp [omin] at h ⊢) <;> omega

/-- Cross-monotonicity of `omax` against an `omin`-order hypothesis. -/
theorem omax_cross' (x y z : Option Nat) (h : omin x y = y) :
    omin (omax x z) (omax y z) = omax y z := by
  cases x <;> cases y <;> cases z <;>
    (try simp [omax, omin] at h ⊢) <;> omega

/-! ### The six operations and four bounds -/

/-- Joining-knowledge on completed records. -/
def joinK (a b : ORecord) : ORecord := ⟨omax a.pos b.pos, omax a.neg b.neg⟩
/-- Meeting-knowledge on completed records. -/
def meetK (a b : ORecord) : ORecord := ⟨omin a.pos b.pos, omin a.neg b.neg⟩
/-- Joining-truth on completed records. -/
def joinT (a b : ORecord) : ORecord := ⟨omax a.pos b.pos, omin a.neg b.neg⟩
/-- Meeting-truth on completed records. -/
def meetT (a b : ORecord) : ORecord := ⟨omin a.pos b.pos, omax a.neg b.neg⟩

/-- The empty completed record — the knowledge bottom, as before. -/
def kBot : ORecord := ⟨some 0, some 0⟩
/-- **The knowledge top: the double infinity.** Told both, unboundedly — the
bound the bare record lacks, restored by the point at infinity. -/
def kTop : ORecord := ⟨none, none⟩
/-- The truth top: infinitely told true, never told false. -/
def tTop : ORecord := ⟨none, some 0⟩
/-- The truth bottom: never told true, infinitely told false. -/
def tBot : ORecord := ⟨some 0, none⟩

/-- The knowledge order on completed records. -/
def OKLE (a b : ORecord) : Prop := ole a.pos b.pos ∧ ole a.neg b.neg

/-- **The completed record is a bounded interlaced bilattice** — adjoining ω
to each axis restores exactly the structure `Record` lacks, so `four_initial`
applies: FOUR maps in, uniquely, with `⊤` landing on the double infinity. -/
def oRecordInterlaced : InterlacedBilattice where
  carrier := ORecord
  meetK := meetK
  joinK := joinK
  kBot := kBot
  kTop := kTop
  meetT := meetT
  joinT := joinT
  tBot := tBot
  tTop := tTop
  joinK_comm := fun _ _ => ext (omax_comm _ _) (omax_comm _ _)
  joinK_assoc := fun _ _ _ => ext (omax_assoc _ _ _) (omax_assoc _ _ _)
  meetK_comm := fun _ _ => ext (omin_comm _ _) (omin_comm _ _)
  absorbK_join := fun _ _ => ext (omax_absorb _ _) (omax_absorb _ _)
  absorbK_meet := fun _ _ => ext (omin_absorb _ _) (omin_absorb _ _)
  joinK_kBot := fun _ => ext (omax_zero _) (omax_zero _)
  meetK_kTop := fun _ => ext (omin_omega _) (omin_omega _)
  joinT_comm := fun _ _ => ext (omax_comm _ _) (omin_comm _ _)
  joinT_assoc := fun _ _ _ => ext (omax_assoc _ _ _) (omin_assoc _ _ _)
  meetT_comm := fun _ _ => ext (omin_comm _ _) (omax_comm _ _)
  absorbT_join := fun _ _ => ext (omax_absorb _ _) (omin_absorb _ _)
  absorbT_meet := fun _ _ => ext (omin_absorb _ _) (omax_absorb _ _)
  joinT_tBot := fun _ => ext (omax_zero _) (omin_omega _)
  meetT_tTop := fun _ => ext (omin_omega _) (omax_zero _)
  meetT_kmono := fun _ _ _ h =>
    ext (omin_cross _ _ _ (congrArg ORecord.pos h)) (omax_cross _ _ _ (congrArg ORecord.neg h))
  joinT_kmono := fun _ _ _ h =>
    ext (omax_cross _ _ _ (congrArg ORecord.pos h)) (omin_cross _ _ _ (congrArg ORecord.neg h))
  meetK_tmono := fun _ _ _ h =>
    ext (omin_cross _ _ _ (congrArg ORecord.pos h)) (omin_cross' _ _ _ (congrArg ORecord.neg h))
  joinK_tmono := fun _ _ _ h =>
    ext (omax_cross _ _ _ (congrArg ORecord.pos h)) (omax_cross' _ _ _ (congrArg ORecord.neg h))

/-- FOUR embeds in the completed record: the unique bounds map is injective
(the four bounds are pairwise distinct), and `⊤` lands on the double
infinity. -/
theorem four_embeds :
    (∀ a b : Charge, boundsMap oRecordInterlaced a = boundsMap oRecordInterlaced b → a = b)
    ∧ boundsMap oRecordInterlaced Charge.kTop = kTop :=
  ⟨boundsMap_injective oRecordInterlaced
      (show kBot ≠ tTop by decide) (show kBot ≠ tBot by decide)
      (show kBot ≠ kTop by decide) (show tTop ≠ tBot by decide)
      (show tTop ≠ kTop by decide) (show tBot ≠ kTop by decide),
    rfl⟩

/-! ### Settling at stage ω — the transfinite-bounded regime witness -/

/-- Finite iteration of an endomap on completed records. -/
def iter (f : ORecord → ORecord) : Nat → ORecord → ORecord
  | 0, r => r
  | k + 1, r => f (iter f k r)

/-- The extended-tally increment: `ω + 1 = ω`. -/
def oadd1 : Option Nat → Option Nat
  | none => none
  | some p => some (p + 1)

/-- The extended order is reflexive. -/
theorem ole_refl (a : Option Nat) : ole a a := by
  cases a with
  | none => exact trivial
  | some x => exact Nat.le_refl x

/-- The increment climbs the extended order. -/
theorem ole_oadd1 (a : Option Nat) : ole a (oadd1 a) := by
  cases a with
  | none => exact trivial
  | some p => exact Nat.le_succ p

/-- The tally increment on completed records: `ω + 1 = ω`. -/
def osuccPos (r : ORecord) : ORecord := ⟨oadd1 r.pos, r.neg⟩

/-- The stage-ω value: infinitely told true, never told false. -/
def omegaPos : ORecord := ⟨none, some 0⟩

theorem osuccPos_infl (r : ORecord) : OKLE r (osuccPos r) :=
  ⟨ole_oadd1 r.pos, ole_refl r.neg⟩

/-- The told-true tally after `k` steps from the empty record. -/
theorem iter_osuccPos_pos (k : Nat) : iter osuccPos k kBot = ⟨some k, some 0⟩ := by
  induction k with
  | zero => rfl
  | succ k ih =>
    show osuccPos (iter osuccPos k kBot) = _
    rw [ih]
    rfl

/-- No finite stage reaches the stage-ω value. -/
theorem iter_ne_omegaPos (k : Nat) : iter osuccPos k kBot ≠ omegaPos := by
  rw [iter_osuccPos_pos]
  intro h
  have hp : (some k : Option Nat) = none := congrArg ORecord.pos h
  exact nomatch hp

/-- Every finite stage sits below the stage-ω value. -/
theorem iter_le_omegaPos (k : Nat) : OKLE (iter osuccPos k kBot) omegaPos := by
  rw [iter_osuccPos_pos]
  exact ⟨trivial, Nat.le_refl 0⟩

/-- The stage-ω value is fixed: `ω + 1 = ω`. -/
theorem osuccPos_omegaPos : osuccPos omegaPos = omegaPos := rfl

/-- The stage-ω value is the least upper bound of the orbit. -/
theorem omegaPos_least (b : ORecord)
    (hb : ∀ k, OKLE (iter osuccPos k kBot) b) : OKLE omegaPos b := by
  refine ⟨?_, (hb 0).2⟩
  cases hbp : b.pos with
  | none => exact trivial
  | some m =>
    have h := (hb (m + 1)).1
    rw [iter_osuccPos_pos, hbp] at h
    exact absurd h (Nat.not_succ_le_self m)

/-- **Settling at stage ω.** On the completed record, the extended tally
increment is inflationary, reaches no fixed point at any finite stage, and
the stage-ω value is its fixed point and the least upper bound of the orbit:
the closure ordinal is exactly ω — the transfinite-bounded regime, inhabited
by a derived carrier. The tower so far: FOUR settles by three
(`Charge.settles_by_three` — the bounded quotient), the bare record never
settles (`Record.record_escapes_strict_finite` — the topless carrier), and
the completed record settles at ω (this theorem — the bound restored at
infinity). -/
theorem settles_at_omega :
    (∀ r, OKLE r (osuccPos r))
    ∧ (∀ k, iter osuccPos k kBot ≠ omegaPos)
    ∧ (∀ k, OKLE (iter osuccPos k kBot) omegaPos)
    ∧ (∀ b, (∀ k, OKLE (iter osuccPos k kBot) b) → OKLE omegaPos b)
    ∧ osuccPos omegaPos = omegaPos :=
  ⟨osuccPos_infl, iter_ne_omegaPos, iter_le_omegaPos, omegaPos_least, rfl⟩

/-- Every extended tally lies beneath ω. -/
theorem ole_none (a : Option Nat) : ole a none := by
  cases a with
  | none => exact trivial
  | some _ => exact trivial

/-- Every completed record lies beneath the double infinity: `kTop` bounds
every ℕ-indexed stage accumulation outright. -/
theorem okle_kTop (r : ORecord) : OKLE r kTop :=
  ⟨ole_none r.pos, ole_none r.neg⟩

end ORecord

/-! ## The germ order — where antisymmetry fails

The aperiodic regime requires a non-antisymmetric carrier
(`Closure.LaxSlackDiagnosis`: lax/strict closure slack is impossible on a
poset). The derived candidate: record sequences under bounded-difference
domination. Two distinct sequences can dominate each other — mutual
domination without identity — so the germ order is a genuine preorder that
is not a partial order. -/

namespace Record

/-- Bounded-difference domination of record sequences: `s` is dominated by
`t` when some fixed record `C` covers the gap at every stage. -/
def DomBy (s t : Nat → Record) : Prop :=
  ∃ C : Record, ∀ n, KLE (s n) (add (t n) C)

theorem domBy_refl (s : Nat → Record) : DomBy s s :=
  ⟨kBot, fun _ => ⟨Nat.le_add_right _ _, Nat.le_add_right _ _⟩⟩

theorem domBy_trans (s t u : Nat → Record)
    (h1 : DomBy s t) (h2 : DomBy t u) : DomBy s u := by
  obtain ⟨C1, hC1⟩ := h1
  obtain ⟨C2, hC2⟩ := h2
  refine ⟨add C2 C1, fun n => ⟨?_, ?_⟩⟩
  · show (s n).pos ≤ (u n).pos + (C2.pos + C1.pos)
    exact Nat.add_assoc (u n).pos C2.pos C1.pos ▸
      Nat.le_trans (hC1 n).1 (Nat.add_le_add_right (hC2 n).1 C1.pos)
  · show (s n).neg ≤ (u n).neg + (C2.neg + C1.neg)
    exact Nat.add_assoc (u n).neg C2.neg C1.neg ▸
      Nat.le_trans (hC1 n).2 (Nat.add_le_add_right (hC2 n).2 C1.neg)

/-- **Antisymmetry fails.** Two distinct record sequences that dominate each
other: the tally climb and its one-step lead. The germ order is a preorder
and not a partial order — the order-deficiency the aperiodic regime needs,
now exhibited on a derived carrier. -/
theorem germ_order_not_antisymm :
    ∃ s t : Nat → Record, DomBy s t ∧ DomBy t s ∧ s ≠ t := by
  refine ⟨fun n => ⟨n, 0⟩, fun n => ⟨n + 1, 0⟩, ?_, ?_, ?_⟩
  · exact ⟨kBot, fun n => ⟨Nat.le_succ n, Nat.le_refl 0⟩⟩
  · exact ⟨⟨1, 0⟩, fun n => ⟨Nat.le_refl (n + 1), Nat.le_refl 0⟩⟩
  · intro h
    have h0 := congrFun h 0
    exact absurd (congrArg Record.pos h0) (by decide)

/-- The germ carrier as a pre-value category: record sequences under
bounded-difference domination, seeded at the constant empty record. -/
def germPreValue : Positing.Value.PreValueCategory where
  carrier := Nat → Record
  le := DomBy
  le_refl := domBy_refl
  le_trans := fun {s t u} h1 h2 => domBy_trans s t u h1 h2
  bot := fun _ => kBot

/-- The tally bump: one more told-true at every stage. -/
def tallyBump (s : Nat → Record) : Nat → Record := fun n => add (s n) ⟨1, 0⟩

/-- The tally bump as a lax closure on the germ carrier: inflationary,
monotone, and Kock–Zöberlein — the double bump is dominated by the single
bump at cost one — without being idempotent anywhere. -/
def tallyLaxClosure : Positing.Value.LaxVClosure germPreValue where
  op := tallyBump
  infl := fun _ => ⟨kBot, fun _ => ⟨Nat.le_succ _, Nat.le_refl _⟩⟩
  mono := fun {s t} h => by
    obtain ⟨C, hC⟩ := h
    refine ⟨C, fun n => ⟨?_, (hC n).2⟩⟩
    show (s n).pos + 1 ≤ (t n).pos + 1 + C.pos
    exact Nat.add_right_comm (t n).pos C.pos 1 ▸ Nat.succ_le_succ (hC n).1
  kz := fun s => ⟨⟨1, 0⟩, fun _ => kle_refl _⟩

/-- **Genuine lax slack, inhabited.** At the seed, the tally bump satisfies
the Kock–Zöberlein inequality while its double differs from its single —
the non-identity 2-cell that `LaxSlackDiagnosis` proves cannot exist on any
antisymmetric carrier, exhibited on the germ preorder. The closure 2-monad's
multiplication carries content for the first time. -/
theorem tally_lax_slack :
    Positing.Value.HasLaxSlack tallyLaxClosure germPreValue.bot := by
  refine ⟨tallyLaxClosure.kz germPreValue.bot, ?_⟩
  intro h
  have h0 := congrFun h 0
  exact absurd (congrArg Record.pos h0) (by decide)

/-! ### The germ carrier is topless yet ω-bounded -/

theorem left_le_max (a b : Nat) : a ≤ max a b := by
  rw [Nat.max_def]
  split
  · assumption
  · exact Nat.le_refl a

theorem right_le_max (a b : Nat) : b ≤ max a b := by
  rw [Nat.max_def]
  split
  · exact Nat.le_refl b
  · next hn => exact Nat.le_of_succ_le (Nat.gt_of_not_le hn)

theorem le_cancel_right : ∀ (a : Nat) {b c : Nat}, b + a ≤ c + a → b ≤ c
  | 0, _, _, h => h
  | a + 1, _, _, h => le_cancel_right a (Nat.le_of_succ_le_succ h)

theorem le_cancel_left {a b c : Nat} (h : a + b ≤ a + c) : b ≤ c := by
  rw [Nat.add_comm a b, Nat.add_comm a c] at h
  exact le_cancel_right a h

/-- The running maximum of `f` over `0..n`. -/
def maxUpTo (f : Nat → Nat) : Nat → Nat
  | 0 => f 0
  | n + 1 => max (maxUpTo f n) (f (n + 1))

theorem le_maxUpTo (f : Nat → Nat) : ∀ n k, k ≤ n → f k ≤ maxUpTo f n
  | 0, 0, _ => Nat.le_refl (f 0)
  | 0, _ + 1, h => absurd h (Nat.not_succ_le_zero _)
  | n + 1, k, h => by
    cases Nat.lt_or_ge k (n + 1) with
    | inl hlt =>
      exact Nat.le_trans (le_maxUpTo f n k (Nat.le_of_lt_succ hlt)) (left_le_max _ _)
    | inr hge =>
      rw [Nat.le_antisymm h hge]
      exact right_le_max (maxUpTo f n) (f (n + 1))

/-- **No ω-sequence of germs escapes.** Every ℕ-indexed family of record
sequences — monotone or not — is dominated by its diagonal running maximum:
the covering constant for the `m`-th member is the running maximum of its
own first `m` stages. The escape that `record_escapes_strict_finite`
exhibits one level down is impossible along ℕ on the germ carrier. -/
theorem germ_omega_family_bounded (g : Nat → Nat → Record) :
    ∃ h : Nat → Record, ∀ m, DomBy (g m) h := by
  refine ⟨fun n => ⟨maxUpTo (fun i => (g i n).pos) n, maxUpTo (fun i => (g i n).neg) n⟩,
    fun m => ⟨⟨maxUpTo (fun n => (g m n).pos) m, maxUpTo (fun n => (g m n).neg) m⟩,
      fun n => ?_⟩⟩
  cases Nat.le_total m n with
  | inl hmn =>
    exact ⟨Nat.le_trans (le_maxUpTo (fun i => (g i n).pos) n m hmn) (Nat.le_add_right _ _),
      Nat.le_trans (le_maxUpTo (fun i => (g i n).neg) n m hmn) (Nat.le_add_right _ _)⟩
  | inr hnm =>
    exact ⟨Nat.le_trans (le_maxUpTo (fun j => (g m j).pos) m n hnm) (Nat.le_add_left _ _),
      Nat.le_trans (le_maxUpTo (fun j => (g m j).neg) m n hnm) (Nat.le_add_left _ _)⟩

/-- **The germ carrier has no top.** Against any candidate `h`, the diagonal
`n ↦ h n + n` outruns every fixed covering constant. With
`germ_omega_family_bounded`: the escape from the germ carrier exists but no
ℕ-indexed family witnesses it. -/
theorem germ_no_top (h : Nat → Record) :
    ∃ g : Nat → Record, ¬ DomBy g h := by
  refine ⟨fun n => ⟨(h n).pos + n, (h n).neg⟩, ?_⟩
  intro ⟨C, hC⟩
  have h1 : (h (C.pos + 1)).pos + (C.pos + 1) ≤ (h (C.pos + 1)).pos + C.pos :=
    (hC (C.pos + 1)).1
  exact absurd (le_cancel_left h1) (Nat.not_succ_le_self C.pos)

/-! ### The aperiodic ⊤-flux -/

/-- Finite iteration of an endomap on record sequences. -/
def iterSeq (f : (Nat → Record) → Nat → Record) : Nat → (Nat → Record) → Nat → Record
  | 0, s => s
  | k + 1, s => f (iterSeq f k s)

/-- The contradiction bump: one more told-true and one more told-false at
every stage. -/
def topBump (s : Nat → Record) : Nat → Record := fun n => add (s n) ⟨1, 1⟩

/-- The contradiction bump as a lax closure on the germ carrier. -/
def topBumpLaxClosure : Positing.Value.LaxVClosure germPreValue where
  op := topBump
  infl := fun _ => ⟨kBot, fun _ => ⟨Nat.le_succ _, Nat.le_succ _⟩⟩
  mono := fun {s t} h => by
    obtain ⟨C, hC⟩ := h
    refine ⟨C, fun n => ⟨?_, ?_⟩⟩
    · show (s n).pos + 1 ≤ (t n).pos + 1 + C.pos
      exact Nat.add_right_comm (t n).pos C.pos 1 ▸ Nat.succ_le_succ (hC n).1
    · show (s n).neg + 1 ≤ (t n).neg + 1 + C.neg
      exact Nat.add_right_comm (t n).neg C.neg 1 ▸ Nat.succ_le_succ (hC n).2
  kz := fun _ => ⟨⟨1, 1⟩, fun _ => kle_refl _⟩

/-- The contradiction bump has lax slack at the seed, like the tally bump. -/
theorem topBump_lax_slack :
    Positing.Value.HasLaxSlack topBumpLaxClosure germPreValue.bot := by
  refine ⟨topBumpLaxClosure.kz germPreValue.bot, ?_⟩
  intro h
  have h0 := congrFun h 0
  exact absurd (congrArg Record.pos h0) (by decide)

theorem iterSeq_topBump (k : Nat) (s : Nat → Record) (n : Nat) :
    iterSeq topBump k s n = ⟨(s n).pos + k, (s n).neg + k⟩ := by
  induction k with
  | zero => rfl
  | succ k ih =>
    show add (iterSeq topBump k s n) ⟨1, 1⟩ = _
    rw [ih]
    rfl

theorem le_min_add (a b k : Nat) : k ≤ min (a + k) (b + k) := by
  rw [Nat.min_def]
  split
  · exact Nat.le_add_left k a
  · exact Nat.le_add_left k b

/-- **The ⊤-flux is order-quiescent and never settles.** Every iterate of
the contradiction bump is germ-equivalent to the seed — the whole orbit is
one point of the germ order — yet consecutive iterates are never equal, and
the contradiction mass at every stage exceeds the iterate count. On the
germ carrier the settling question splits: the order settles instantly, the
data never does, and the gap is an unbounded held-⊤ flux inside a single
germ. -/
theorem topBump_flux (s : Nat → Record) :
    (∀ k, DomBy (iterSeq topBump k s) s ∧ DomBy s (iterSeq topBump k s))
    ∧ (∀ k, iterSeq topBump (k + 1) s ≠ iterSeq topBump k s)
    ∧ (∀ k n, k ≤ topMass (iterSeq topBump k s n)) := by
  refine ⟨fun k => ⟨⟨⟨k, k⟩, fun n => ?_⟩, ⟨kBot, fun n => ?_⟩⟩, fun k h => ?_, fun k n => ?_⟩
  · rw [iterSeq_topBump]; exact kle_refl _
  · rw [iterSeq_topBump]
    exact ⟨Nat.le_add_right _ _, Nat.le_add_right _ _⟩
  · have h0 := congrArg Record.pos (congrFun h 0)
    rw [iterSeq_topBump, iterSeq_topBump] at h0
    exact absurd (le_cancel_left (Nat.le_of_eq h0)) (Nat.not_succ_le_self k)
  · rw [iterSeq_topBump]
    exact le_min_add (s n).pos (s n).neg k

end Record
